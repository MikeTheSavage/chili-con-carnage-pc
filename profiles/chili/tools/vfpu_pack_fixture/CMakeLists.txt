cmake_minimum_required(VERSION 3.20)
project(ChiliVfpuPackFixture LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(DEFAULT_REPO "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(REPOSITORY_ROOT "${DEFAULT_REPO}" CACHE PATH "PSPRecomp checkout containing the live core library")
set(FIXTURE_OUTPUT_DIR "${REPOSITORY_ROOT}/profiles/chili/analysis/vfpu_pack_fixture" CACHE PATH "Generated fixture output")
if(NOT EXISTS "${FIXTURE_OUTPUT_DIR}/generated.cpp")
    message(FATAL_ERROR "Run generate_fixture.ps1 with the live psp_recomp before configuring this test")
endif()
add_executable(chili_vfpu_pack_test pack_fixture_test.cpp "${FIXTURE_OUTPUT_DIR}/generated.cpp")
target_include_directories(chili_vfpu_pack_test PRIVATE "${REPOSITORY_ROOT}/include")
target_link_libraries(chili_vfpu_pack_test PRIVATE "${REPOSITORY_ROOT}/out/framework/Release/psprecomp_core.lib")
if(MSVC)
    target_compile_options(chili_vfpu_pack_test PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME chili_vfpu_pack COMMAND chili_vfpu_pack_test)
